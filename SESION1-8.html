Módulos en Python

Un módulo es un archivo que contiene definiciones y declaraciones de Python. El nombre del archivo es el nombre del módulo con el sufijo .py.

Crear un módulo:

Crea un archivo llamado saludos.py con el siguiente contenido:

# saludos.py
def saludar(nombre):
    """Saluda a una persona por su nombre."""
    return f"¡Hola, {nombre}! ¿Cómo estás?"

PI = 3.14159

class Persona:
    def __init__(self, nombre, edad):
        self.nombre = nombre
        self.edad = edad
    
    def presentarse(self):
        return f"Soy {self.nombre} y tengo {self.edad} años."

Usar un módulo:

# Importar todo el módulo
import saludos

# Usar funciones, variables y clases del módulo
print(saludos.saludar("Juan"))
print(f"El valor de PI es: {saludos.PI}"

persona = saludos.Persona("Ana", 25)
print(persona.presentarse())

# Importar elementos específicos
from saludos import saludar, PI

# Ahora podemos usarlos directamente
print(saludar("Carlos"))
print(PI)

# Importar con un alias
import saludos as s
from saludos import Persona as P

print(s.saludar("María")
persona2 = P("Pedro", 30)

Paquetes en Python

Un paquete es una forma de organizar módulos relacionados en una jerarquía de directorios.

Estructura de un paquete:

mi_paquete/
├── __init__.py
├── modulo1.py
├── modulo2.py
└── subpaquete/
    ├── __init__.py
    └── modulo3.py

Crear un paquete:

1. Crea una carpeta llamada mi_paquete

2. Dentro de la carpeta, crea un archivo __init__.py (puede estar vacío o contener código de inicialización)

Ejemplo de un paquete simple:

mi_paquete/__init__.py:

# Este archivo hace que Python trate el directorio como un paquete
__version__ = '1.0.0'

# Podemos definir qué se importa con 'from paquete import *'
__all__ = ['modulo1', 'modulo2']

mi_paquete/modulo1.py:

def hola():
    return "¡Hola desde modulo1!"

mi_paquete/modulo2.py:

def despedida():
    return "¡Adiós desde modulo2!"

Usar el paquete:

# Importar todo el paquete
import mi_paquete

# Usar módulos del paquete
print(mi_paquete.modulo1.hola())
print(mi_paquete.modulo2.despedida())

# Importar módulos específicos
from mi_paquete import modulo1, modulo2

# Importar funciones específicas
from mi_paquete.modulo1 import hola
from mi_paquete.modulo2 import despedida as adios

Paquetes anidados:

Puedes tener paquetes dentro de paquetes. Por ejemplo, crea una carpeta subpaquete dentro de mi_paquete con su propio __init__.py y módulos.

Gestión de Paquetes con pip

pip es el sistema de gestión de paquetes de Python. Te permite instalar y administrar paquetes de terceros.

Comandos básicos de pip:

# Instalar un paquete
pip install nombre_del_paquete

# Instalar una versión específica
pip install nombre_del_paquete==1.2.3

# Actualizar un paquete
pip install --upgrade nombre_del_paquete

# Desinstalar un paquete
pip uninstall nombre_del_paquete

# Listar paquetes instalados
pip list

# Ver información de un paquete instalado
pip show nombre_del_paquete

# Guardar dependencias en un archivo
pip freeze > requirements.txt

# Instalar dependencias desde un archivo
pip install -r requirements.txt

Entornos virtuales:

Es una buena práctica usar entornos virtuales para aislar las dependencias de cada proyecto.

# Crear un entorno virtual
python -m venv mi_entorno

# Activar el entorno (Windows)
mi_entorno\Scripts\activate

# Activar el entorno (macOS/Linux)
source mi_entorno/bin/activate

# Desactivar el entorno (cualquier sistema)
deactivate

gestor_tareas/tareas.py

class Tarea:
    def __init__(self, titulo, descripcion='', completada=False):
        self.titulo = titulo
        self.descripcion = descripcion
        self.completada = completada
        self.fecha_creacion = datetime.datetime.now()
    
    def marcar_completada(self):
        self.completada = True
    
    def __str__(self):
        estado = "✓" if self.completada else " "
        return f"[{estado}] {self.titulo}"


class GestorTareas:
    def __init__(self):
        self.tareas = []
    
    def agregar_tarea(self, titulo, descripcion=''):
        tarea = Tarea(titulo, descripcion)
        self.tareas.append(tarea)
        return tarea
    
    def listar_tareas(self, solo_pendientes=False):
        if solo_pendientes:
            return [t for t in self.tareas if not t.completada]
        return self.tareas
    
    def marcar_como_completada(self, indice):
        if 0 <= indice < len(self.tareas):
            self.tareas[indice].marcar_completada()
            return True
        return False

gestor_tareas/almacenamiento.py

import json
from datetime import datetime

class AlmacenamientoTareas:
    def __init__(self, archivo='tareas.json'):
        self.archivo = archivo
    
    def guardar(self, gestor_tareas):
        datos = []
        for tarea in gestor_tareas.tareas:
            datos.append({
                'titulo': tarea.titulo,
                'descripcion': tarea.descripcion,
                'completada': tarea.completada,
                'fecha_creacion': tarea.fecha_creacion.isoformat()
            })
        
        with open(self.archivo, 'w') as f:
            json.dump(datos, f, indent=2)
    
    def cargar(self):
        try:
            with open(self.archivo, 'r') as f:
                datos = json.load(f)
            
            gestor = GestorTareas()
            for tarea_data in datos:
                tarea = Tarea(
                    tarea_data['titulo'],
                    tarea_data.get('descripcion', ''),
                    tarea_data.get('completada', False)
                )
                tarea.fecha_creacion = datetime.fromisoformat(tarea_data['fecha_creacion'])
                gestor.tareas.append(tarea)
            
            return gestor
        except FileNotFoundError:
            return GestorTareas()

gestor_tareas/interfaz/consola.py

from ..tareas import GestorTareas

class InterfazConsola:
    def __init__(self, gestor_tareas):
        self.gestor = gestor_tareas
    
    def mostrar_menu(self):
        print("\n--- Gestor de Tareas ---")
        print("1. Ver tareas")
        print("2. Agregar tarea")
        print("3. Marcar tarea como completada")
        print("4. Salir")
    
    def ejecutar(self):
        while True:
            self.mostrar_menu()
            opcion = input("\nSelecciona una opción: ")
            
            if opcion == '1':
                self.mostrar_tareas()
            elif opcion == '2':
                self.agregar_tarea()
            elif opcion == '3':
                self.marcar_como_completada()
            elif opcion == '4':
                print("¡Hasta luego!")
                break
            else:
                print("Opción no válida. Intenta de nuevo.")
    
    def mostrar_tareas(self):
        print("\n--- Tareas ---")
        tareas = self.gestor.listar_tareas()
        if not tareas:
            print("No hay tareas.")
            return
        
        for i, tarea in enumerate(tareas, 1):
            print(f"{i}. {tarea}")
    
    def agregar_tarea(self):
        print("\n--- Nueva Tarea ---")
        titulo = input("Título: ")
        descripcion = input("Descripción (opcional): ")
        self.gestor.agregar_tarea(titulo, descripcion)
        print("¡Tarea agregada!")
    
    def marcar_como_completada(self):
        self.mostrar_tareas()
        tareas = self.gestor.listar_tareas(solo_pendientes=True)
        
        if not tareas:
            print("No hay tareas pendientes.")
            return
        
        try:
            indice = int(input("Número de tarea a marcar como completada: ")) - 1
            if 0 <= indice < len(tareas):
                tarea = tareas[indice]
                tarea.marcar_completada()
                print(f"Tarea '{tarea.titulo}' marcada como completada.")
            else:
                print("Número de tarea inválido.")
        except ValueError:
            print("Por favor ingresa un número válido.")

main.py

from gestor_tareas.almacenamiento import AlmacenamientoTareas
from gestor_tareas.interfaz.consola import InterfazConsola

def main():
    # Inicializar el almacenamiento
    almacenamiento = AlmacenamientoTareas()
    
    # Cargar tareas existentes
    gestor = almacenamiento.cargar()
    
    # Iniciar la interfaz de usuario
    interfaz = InterfazConsola(gestor)
    
    try:
        interfaz.ejecutar()
    finally:
        # Guardar las tareas al salir
        almacenamiento.guardar(gestor)

if __name__ == "__main__":
    main()

Instrucciones para ejecutar el proyecto:

  2. Copia el código correspondiente a cada archivo
  3. Asegúrate de tener Python instalado
  4. Ejecuta el programa con: python main.py

Resumen

Módulos

  • Archivos .py con código Python
  • Pueden contener funciones, clases y variables
  • Se importan con import modulo o from modulo import algo

Paquetes

  • Directorios con un archivo __init__.py
  • Organizan módulos relacionados
  • Pueden contener subpaquetes

PIP

  • Sistema de gestión de paquetes
  • pip install paquete para instalar
  • requirements.txt para dependencias
